-- HeartMed - Script de base de datos para SQLite

-- Habilitar el soporte de claves foráneas en SQLite
PRAGMA foreign_keys = ON;

-- Limpieza de tablas previas (en orden inverso a sus relaciones)
DROP TABLE IF EXISTS logs_auditoria;
DROP TABLE IF EXISTS usuarios;
DROP TABLE IF EXISTS rol_permiso;
DROP TABLE IF EXISTS permisos;
DROP TABLE IF EXISTS roles;
DROP TABLE IF EXISTS pasos_protocolo;
DROP TABLE IF EXISTS protocolos;

-- ==========================================
-- ESTRUCTURA DE TABLAS
-- ==========================================

-- Tabla de roles
CREATE TABLE roles (
    id_rol        INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre_rol    TEXT NOT NULL UNIQUE,
    descripcion   TEXT NULL
);

-- Tabla de permisos
CREATE TABLE permisos (
    id_permiso      INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre_permiso  TEXT NOT NULL,
    cod_acceso      TEXT NOT NULL UNIQUE
);

-- Relación entre roles y permisos
CREATE TABLE rol_permiso (
    id_rol      INTEGER NOT NULL,
    id_permiso  INTEGER NOT NULL,
    PRIMARY KEY (id_rol, id_permiso),
    FOREIGN KEY (id_rol) REFERENCES roles (id_rol) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (id_permiso) REFERENCES permisos (id_permiso) ON DELETE CASCADE ON UPDATE CASCADE
);

-- Tabla de usuarios
CREATE TABLE usuarios (
    id_usuario       INTEGER PRIMARY KEY AUTOINCREMENT,
    id_rol           INTEGER NOT NULL,
    nombre           TEXT NOT NULL,
    correo           TEXT NOT NULL UNIQUE,
    contrasena_hash  TEXT NOT NULL,
    activo           INTEGER NOT NULL DEFAULT 1, -- 1 = TRUE, 0 = FALSE
    fecha_creacion   TEXT NOT NULL DEFAULT (DATETIME('now')),
    FOREIGN KEY (id_rol) REFERENCES roles (id_rol) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- Tabla de auditoría
CREATE TABLE logs_auditoria (
    id_log              INTEGER PRIMARY KEY AUTOINCREMENT,
    id_usuario          INTEGER NOT NULL,
    accion              TEXT NOT NULL,
    tabla_afectada      TEXT NOT NULL,
    id_registro         INTEGER NULL,
    detalle_cambio      TEXT NULL,
    fecha_modificacion  TEXT NOT NULL DEFAULT (DATETIME('now')),
    FOREIGN KEY (id_usuario) REFERENCES usuarios (id_usuario) ON DELETE RESTRICT ON UPDATE CASCADE
);

-- Protocolo de emergencias
CREATE TABLE protocolos (
    id_protocolo     INTEGER PRIMARY KEY AUTOINCREMENT,
    tipo_emergencia  TEXT NOT NULL CHECK(tipo_emergencia IN ('TRAUMATISMO', 'HEMORRAGIA', 'INTOXICACION', 'QUEMADURA', 'OTRO')),
    nombre           TEXT NOT NULL,
    CONSTRAINT uk_protocolos_tipo UNIQUE (tipo_emergencia)
);

-- Pasos del protocolo
CREATE TABLE pasos_protocolo (
    id_paso       INTEGER PRIMARY KEY AUTOINCREMENT,
    id_protocolo  INTEGER NOT NULL,
    numero        INTEGER NOT NULL,
    descripcion   TEXT NOT NULL,
    UNIQUE (id_protocolo, numero),
    FOREIGN KEY (id_protocolo) REFERENCES protocolos (id_protocolo) ON DELETE CASCADE ON UPDATE CASCADE
);
